Notification trong Godot

Mọi Object trong Godot đều triển khai phương thức _notification. Mục đích của nó là cho phép Object phản hồi nhiều callback cấp engine có thể liên quan đến nó. Ví dụ, nếu engine bảo một CanvasItem "vẽ", nó sẽ gọi _notification(NOTIFICATION_DRAW).

Một số notification này, như draw, rất hữu ích để ghi đè trong script. Hữu ích đến mức Godot công khai nhiều notification trong số đó bằng các hàm chuyên dụng:

  • _ready(): NOTIFICATION_READY

  • _enter_tree(): NOTIFICATION_ENTER_TREE

  • _exit_tree(): NOTIFICATION_EXIT_TREE

  • _process(delta): NOTIFICATION_PROCESS

  • _physics_process(delta): NOTIFICATION_PHYSICS_PROCESS

  • _draw(): NOTIFICATION_DRAW

Điều mà người dùng có thể không nhận ra là notification tồn tại cho cả những kiểu khác ngoài Node, ví dụ:

  • Object::NOTIFICATION_POSTINITIALIZE: một callback được kích hoạt trong quá trình khởi tạo đối tượng. Script không truy cập được.

  • Object::NOTIFICATION_PREDELETE: một callback được kích hoạt trước khi engine xóa một Object, tức là một "hàm hủy" (destructor).

Và nhiều callback có tồn tại trong Node không có phương thức chuyên dụng nào, nhưng vẫn khá hữu ích.

  • Node::NOTIFICATION_PARENTED: một callback được kích hoạt mỗi khi bạn thêm một node con vào một node khác.

  • Node::NOTIFICATION_UNPARENTED: một callback được kích hoạt mỗi khi bạn gỡ một node con khỏi một node khác.

Phương thức chung _notification() cho phép truy cập tất cả các notification tùy chỉnh này.

Ghi chú

Các phương thức trong tài liệu được đánh dấu là "virtual" cũng được thiết kế để script ghi đè.

Một ví dụ kinh điển là phương thức _init trong Object. Dù nó không có notification NOTIFICATION_* tương đương, engine vẫn gọi phương thức này. Hầu hết các ngôn ngữ (trừ C#) dựa vào nó như một hàm khởi tạo (constructor).

Vậy, khi nào bạn nên dùng từng notification hoặc hàm virtual này?

_process so với _physics_process so với *_input

Dùng _process() khi bạn cần delta time giữa các khung hình phụ thuộc vào framerate. Nếu code cập nhật dữ liệu đối tượng cần cập nhật thường xuyên nhất có thể, thì đây là chỗ phù hợp. Các bước kiểm tra logic lặp lại và việc lưu đệm dữ liệu thường thực thi ở đây, nhưng điều đó còn tùy vào tần suất cần cập nhật các phép đánh giá. Nếu chúng không cần thực thi mỗi khung hình, thì triển khai một vòng lặp Timer-timeout là một lựa chọn khác.

# Allows for recurring operations that don't trigger script logic
# every frame (or even every fixed frame).
func _ready():
    var timer = Timer.new()
    timer.autostart = true
    timer.wait_time = 0.5
    add_child(timer)
    timer.timeout.connect(func():
        print("This block runs every 0.5 seconds")
    )

using Godot;

public partial class MyNode : Node
{
    // Allows for recurring operations that don't trigger script logic
    // every frame (or even every fixed frame).
    public override void _Ready()
    {
        var timer = new Timer();
        timer.Autostart = true;
        timer.WaitTime = 0.5;
        AddChild(timer);
        timer.Timeout += () => GD.Print("This block runs every 0.5 seconds");
    }
}

using namespace godot;

class MyNode : public Node {
    GDCLASS(MyNode, Node)

public:
    // Allows for recurring operations that don't trigger script logic
    // every frame (or even every fixed frame).
    virtual void _ready() override {
        Timer *timer = memnew(Timer);
        timer->set_autostart(true);
        timer->set_wait_time(0.5);
        add_child(timer);
        timer->connect("timeout", callable_mp(this, &MyNode::run));
    }

    void run() {
        UtilityFunctions::print("This block runs every 0.5 seconds.");
    }
};

Dùng _physics_process() khi bạn cần delta time giữa các khung hình không phụ thuộc vào framerate. Nếu code cần cập nhật đều đặn theo thời gian, bất kể thời gian trôi nhanh hay chậm, thì đây là chỗ phù hợp. Các thao tác kinematic và biến đổi transform của đối tượng lặp lại nên thực thi ở đây.

Dù có thể làm được, để đạt hiệu năng tốt nhất, bạn nên tránh kiểm tra input trong các callback này. _process() và _physics_process() sẽ kích hoạt ở mọi cơ hội (mặc định chúng không "nghỉ"). Ngược lại, các callback *_input() chỉ kích hoạt ở những khung hình mà engine thực sự phát hiện có input.

Bạn vẫn có thể kiểm tra các input action trong các callback input như thường. Nếu muốn dùng delta time, bạn có thể lấy nó từ các phương thức delta time liên quan khi cần.

# Called every frame, even when the engine detects no input.
func _process(delta):
    if Input.is_action_just_pressed("ui_select"):
        print(delta)

# Called during every input event.
func _unhandled_input(event):
    match event.get_class():
        "InputEventKey":
            if Input.is_action_just_pressed("ui_accept"):
                print(get_process_delta_time())

using Godot;

public partial class MyNode : Node
{

    // Called every frame, even when the engine detects no input.
    public void _Process(double delta)
    {
        if (Input.IsActionJustPressed("ui_select"))
        {
            GD.Print(delta);
        }
    }

    // Called during every input event. Equally true for _input().
    public void _UnhandledInput(InputEvent @event)
    {
        switch (@event)
        {
            case InputEventKey:
                if (Input.IsActionJustPressed("ui_accept"))
                {
                    GD.Print(GetProcessDeltaTime());
                }
                break;
        }
    }

}

using namespace godot;

class MyNode : public Node {
    GDCLASS(MyNode, Node)

public:
    // Called every frame, even when the engine detects no input.
    virtual void _process(double p_delta) override {
        if (Input::get_singleton->is_action_just_pressed("ui_select")) {
            UtilityFunctions::print(p_delta);
        }
    }

    // Called during every input event. Equally true for _input().
    virtual void _unhandled_input(const Ref<InputEvent> &p_event) override {
        Ref<InputEventKey> key_event = event;
        if (key_event.is_valid() && Input::get_singleton->is_action_just_pressed("ui_accept")) {
            UtilityFunctions::print(get_process_delta_time());
        }
    }
};

_init so với khởi tạo so với export

Nếu script tự khởi tạo cây con node của riêng nó, mà không có scene, thì code đó nên thực thi trong _init(). Các khởi tạo khác cho thuộc tính hoặc không phụ thuộc SceneTree cũng nên chạy ở đây.

Ghi chú

Tương đương trong C# của phương thức _init() trong GDScript là hàm khởi tạo (constructor).

_init() kích hoạt trước _enter_tree() hoặc _ready(), nhưng sau khi script tạo và khởi tạo các thuộc tính của nó. Khi tạo instance một scene, giá trị thuộc tính sẽ được thiết lập theo trình tự sau:

  1. Gán giá trị ban đầu: thuộc tính được gán giá trị khởi tạo của nó, hoặc giá trị mặc định nếu không được chỉ định. Nếu có setter, setter không được dùng.

  2. _init() gán giá trị: giá trị thuộc tính được thay thế bởi mọi phép gán thực hiện trong _init(), và kích hoạt setter.

  3. Gán giá trị export: giá trị của thuộc tính được export lại được thay thế bởi bất kỳ giá trị nào được đặt trong Inspector, và kích hoạt setter.

# test is initialized to "one", without triggering the setter.
@export var test: String = "one":
    set(value):
        test = value + "!"

func _init():
    # Triggers the setter, changing test's value from "one" to "two!".
    test = "two"

# If you set test to "three" from the Inspector, it would trigger
# the setter, changing test's value from "two!" to "three!".

using Godot;

public partial class MyNode : Node
{
    private string _test = "one";

    [Export]
    public string Test
    {
        get { return _test; }
        set { _test = $"{value}!"; }
    }

    public MyNode()
    {
        // Triggers the setter, changing _test's value from "one" to "two!".
        Test = "two";
    }

    // If you set Test to "three" in the Inspector, it would trigger
    // the setter, changing _test's value from "two!" to "three!".
}

using namespace godot;

class MyNode : public Node {
    GDCLASS(MyNode, Node)

    String test = "one";

protected:
    static void _bind_methods() {
        ClassDB::bind_method(D_METHOD("get_test"), &MyNode::get_test);
        ClassDB::bind_method(D_METHOD("set_test", "test"), &MyNode::set_test);
        ADD_PROPERTY(PropertyInfo(Variant::STRING, "test"), "set_test", "get_test");
    }

public:
    String get_test() { return test; }
    void set_test(String p_test) { test = p_test + "!"; }

    MyNode() {
        // Triggers the setter, changing _test's value from "one" to "two!".
        set_test("two");
    }

    // If you set test to "three" in the Inspector, it would trigger
    // the setter, changing test's value from "two!" to "three!".
};

Do đó, tạo instance một script so với một scene có thể ảnh hưởng đến cả quá trình khởi tạo lẫn số lần engine gọi setter.

_ready so với _enter_tree so với NOTIFICATION_PARENTED

Khi tạo instance một scene được kết nối với scene thực thi đầu tiên, Godot sẽ tạo instance các node đi xuống cây (gọi _init()) và xây dựng cây đi xuống từ gốc. Điều này khiến các lời gọi _enter_tree() lan dần xuống cây. Khi cây đã hoàn chỉnh, các node lá gọi _ready. Một node sẽ gọi phương thức này khi tất cả các node con đã gọi xong phương thức của chúng. Điều này tạo ra một chuỗi lan ngược đi lên trở lại gốc của cây.

Khi tạo instance một script hoặc một scene độc lập, các node không được thêm vào SceneTree ngay khi được tạo, nên không có callback _enter_tree() nào kích hoạt. Thay vào đó, chỉ có lời gọi _init() xảy ra. Khi scene được thêm vào SceneTree, các lời gọi _enter_tree() và _ready() mới xảy ra.

Nếu bạn cần kích hoạt hành vi xảy ra khi node được gắn làm con của node khác, bất kể việc đó diễn ra trong scene chính/đang hoạt động hay không, bạn có thể dùng notification PARENTED. Ví dụ, đây là một đoạn code kết nối phương thức của một node với một signal tùy chỉnh trên node cha mà không bị lỗi. Hữu ích với các node tập trung vào dữ liệu có thể được tạo lúc chạy.

extends Node

var parent_cache

func connection_check():
    return parent_cache.has_user_signal("interacted_with")

func _notification(what):
    match what:
        NOTIFICATION_PARENTED:
            parent_cache = get_parent()
            if connection_check():
                parent_cache.interacted_with.connect(_on_parent_interacted_with)
        NOTIFICATION_UNPARENTED:
            if connection_check():
                parent_cache.interacted_with.disconnect(_on_parent_interacted_with)

func _on_parent_interacted_with():
    print("I'm reacting to my parent's interaction!")

using Godot;

public partial class MyNode : Node
{
    private Node _parentCache;

    public bool ConnectionCheck()
    {
        return _parentCache.HasUserSignal("InteractedWith");
    }

    public override void _Notification(int what)
    {
        switch ((long)what)
        {
            case NotificationParented:
                _parentCache = GetParent();
                if (ConnectionCheck())
                {
                    _parentCache.Connect("InteractedWith", Callable.From(OnParentInteractedWith));
                }
                break;
            case NotificationUnparented:
                if (ConnectionCheck())
                {
                    _parentCache.Disconnect("InteractedWith", Callable.From(OnParentInteractedWith));
                }
                break;
        }
    }

    private void OnParentInteractedWith()
    {
        GD.Print("I'm reacting to my parent's interaction!");
    }
}

using namespace godot;

class MyNode : public Node {
    GDCLASS(MyNode, Node)

    Node *parent_cache = nullptr;

    void on_parent_interacted_with() {
        UtilityFunctions::print("I'm reacting to my parent's interaction!");
    }

public:
    void connection_check() {
        return parent_cache->has_user_signal("interacted_with");
    }

    void _notification(int p_what) {
        switch (p_what) {
            case NOTIFICATION_PARENTED:
                parent_cache = get_parent();
                if (connection_check()) {
                    parent_cache->connect("interacted_with", callable_mp(this, &MyNode::on_parent_interacted_with));
                }
                break;
            case NOTIFICATION_UNPARENTED:
                if (connection_check()) {
                    parent_cache->disconnect("interacted_with", callable_mp(this, &MyNode::on_parent_interacted_with));
                }
                break;
        }
    }
};
